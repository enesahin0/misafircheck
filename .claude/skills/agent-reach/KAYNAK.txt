CLI bulut konteynerinde her oturumda yeniden kurulmalı (konteyner geçici):
  git clone --depth 1 https://github.com/Panniantong/Agent-Reach.git /tmp/ar
  python3 -m venv ~/.agent-reach-venv && ~/.agent-reach-venv/bin/pip install /tmp/ar
  ln -sf ~/.agent-reach-venv/bin/agent-reach /usr/local/bin/agent-reach && agent-reach install --env=auto --system
